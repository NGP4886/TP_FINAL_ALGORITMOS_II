DROP DATABASE IF EXISTS coworkingdb;
CREATE DATABASE coworkingdb;
USE coworkingdb;


CREATE TABLE Clientes (
    id_cliente INT AUTO_INCREMENT PRIMARY KEY,
    dni VARCHAR(20) NOT NULL UNIQUE,
    nombre VARCHAR(50) NOT NULL,
    apellido VARCHAR(50) NOT NULL,
    telefono VARCHAR(20),
    email VARCHAR(100),
    activo BOOLEAN DEFAULT TRUE
);

CREATE TABLE TiposRecurso (
    id_tipo INT AUTO_INCREMENT PRIMARY KEY,
    descripcion VARCHAR(50) NOT NULL UNIQUE
);

-- 3. Tabla Recursos central
CREATE TABLE Recursos (
    id_recurso INT AUTO_INCREMENT PRIMARY KEY,
    id_tipo INT NOT NULL,
    nombre VARCHAR(100) NOT NULL,
    ubicacion VARCHAR(100) NOT NULL,
    capacidad INT NOT NULL DEFAULT 1,
    precio_por_hora DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    estado ENUM('Disponible', 'Mantenimiento', 'Inactivo') DEFAULT 'Disponible',
    FOREIGN KEY (id_tipo) REFERENCES TiposRecurso(id_tipo) ON DELETE RESTRICT
);

-- 4. NUEVO: Tabla catálogo de Adicionales
CREATE TABLE Adicionales (
    id_adicional INT AUTO_INCREMENT PRIMARY KEY,
    id_tipo_recurso INT NOT NULL, -- ACÁ ESTÁ LA VALIDACIÓN: Este adicional solo aplica a este tipo
    descripcion VARCHAR(100) NOT NULL, -- Ej: "Proyector", "Barra Libre", "Monitor Extra"
    costo_extra DECIMAL(10,2) DEFAULT 0.00, -- Si el adicional cobra un recargo
    FOREIGN KEY (id_tipo_recurso) REFERENCES TiposRecurso(id_tipo) ON DELETE CASCADE
);

-- 5. NUEVO: Tabla Intermedia (Qué recurso tiene qué adicional)
CREATE TABLE Recursos_Adicionales (
    id_recurso INT,
    id_adicional INT,
    PRIMARY KEY (id_recurso, id_adicional),
    FOREIGN KEY (id_recurso) REFERENCES Recursos(id_recurso) ON DELETE CASCADE,
    FOREIGN KEY (id_adicional) REFERENCES Adicionales(id_adicional) ON DELETE CASCADE
);

-- 6. Tabla Reservas
CREATE TABLE Reservas (
    id_reserva INT AUTO_INCREMENT PRIMARY KEY,
    id_cliente INT NOT NULL,
    id_recurso INT NOT NULL,
    fecha_reserva DATE NOT NULL,
    hora_inicio TIME NOT NULL,
    hora_fin TIME NOT NULL,
    cantidad_personas INT NOT NULL DEFAULT 1,
    costo_total DECIMAL(10,2) DEFAULT 0.00,
    estado ENUM('Activa', 'Cancelada', 'Finalizada') DEFAULT 'Activa',
    fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_cliente) REFERENCES Clientes(id_cliente) ON DELETE RESTRICT,
    FOREIGN KEY (id_recurso) REFERENCES Recursos(id_recurso) ON DELETE RESTRICT
);


DELIMITER //

-- =========================================================
-- FUNCIÓN: Verificar Disponibilidad (Filtro de Solapamiento)
-- =========================================================
CREATE FUNCTION fn_verificar_disponibilidad(
    p_id_recurso INT, 
    p_fecha DATE, 
    p_hora_inicio TIME, 
    p_hora_fin TIME,
    p_id_reserva_excluir INT -- Para ignorar la reserva actual si estamos modificando
) RETURNS BOOLEAN
DETERMINISTIC
BEGIN
    DECLARE v_existe INT;
    DECLARE v_conflictos INT;
    
    -- Verificar existencia real del recurso
    SELECT COUNT(*) INTO v_existe FROM Recursos WHERE id_recurso = p_id_recurso;
    IF v_existe = 0 THEN 
        RETURN FALSE; 
    END IF;
    
    -- Buscar superposiciones en reservas activas para ese mismo día
    SELECT COUNT(*) INTO v_conflictos
    FROM Reservas
    WHERE id_recurso = p_id_recurso 
      AND fecha_reserva = p_fecha
      AND estado = 'Activa'
      AND id_reserva != IFNULL(p_id_reserva_excluir, 0)
      AND (
          (p_hora_inicio >= hora_inicio AND p_hora_inicio < hora_fin) OR
          (p_hora_fin > hora_inicio AND p_hora_fin <= hora_fin) OR
          (p_hora_inicio <= hora_inicio AND p_hora_fin >= hora_fin)
      );
      
    RETURN IF(v_conflictos > 0, FALSE, TRUE);
END //


-- =========================================================
-- PROCEDIMIENTO 1: Alta de Reserva
-- =========================================================
DELIMITER //
CREATE PROCEDURE sp_alta_reserva(
    IN p_id_cliente INT,
    IN p_id_recurso INT,
    IN p_fecha DATE,
    IN p_hora_inicio TIME,
    IN p_hora_fin TIME,
    IN p_cantidad_personas INT
)
BEGIN
    DECLARE v_recurso_estado VARCHAR(20);
    DECLARE v_capacidad INT;
    DECLARE v_precio_hora DECIMAL(10,2);
    DECLARE v_horas DECIMAL(10,2);
    DECLARE v_costo_total DECIMAL(10,2);
    
    -- 1. Validar Cliente
    IF NOT EXISTS(SELECT 1 FROM Clientes WHERE id_cliente = p_id_cliente AND activo = TRUE) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El cliente no existe o se encuentra inactivo.';
    END IF;

    -- 2. Validar Recurso y Capacidad
    SELECT estado, capacidad, precio_por_hora INTO v_recurso_estado, v_capacidad, v_precio_hora 
    FROM Recursos WHERE id_recurso = p_id_recurso;
    
    IF v_recurso_estado IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El recurso seleccionado no existe.';
    ELSEIF v_recurso_estado != 'Disponible' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El recurso seleccionado no se encuentra disponible.';
    END IF;

    IF p_cantidad_personas > v_capacidad THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La cantidad de personas supera la capacidad máxima del recurso.';
    END IF;

    -- 3. Validar Tiempos y Fechas
    IF p_hora_inicio >= p_hora_fin THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La hora de inicio debe ser anterior a la hora de fin.';
    END IF;

    IF p_fecha < CURDATE() OR (p_fecha = CURDATE() AND p_hora_inicio <= CURTIME()) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'No se pueden registrar reservas en fechas u horarios que ya pasaron.';
    END IF;
    
    -- 4. Validar Solapamiento
    IF fn_verificar_disponibilidad(p_id_recurso, p_fecha, p_hora_inicio, p_hora_fin, NULL) = FALSE THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El recurso ya se encuentra reservado en el horario seleccionado.';
    END IF;
    
    -- 5. Calcular costo básico e insertar
    SET v_horas = TIMESTAMPDIFF(MINUTE, p_hora_inicio, p_hora_fin) / 60.0;
    SET v_costo_total = v_precio_hora * v_horas;
    
    INSERT INTO Reservas (id_cliente, id_recurso, fecha_reserva, hora_inicio, hora_fin, cantidad_personas, costo_total)
    VALUES (p_id_cliente, p_id_recurso, p_fecha, p_hora_inicio, p_hora_fin, p_cantidad_personas, v_costo_total);
END //
DELIMITER ;

-- =========================================================
-- PROCEDIMIENTO 2: Modificar Reserva
-- =========================================================
DELIMITER //
CREATE PROCEDURE sp_modificar_reserva(
    IN p_id_reserva INT,
    IN p_nueva_fecha DATE,
    IN p_nueva_hora_inicio TIME,
    IN p_nueva_hora_fin TIME,
    IN p_nueva_cantidad_personas INT
)
BEGIN
    DECLARE v_id_recurso INT;
    DECLARE v_estado VARCHAR(20);
    DECLARE v_capacidad INT;
    DECLARE v_precio_hora DECIMAL(10,2);
    DECLARE v_horas DECIMAL(10,2);
    DECLARE v_nuevo_costo DECIMAL(10,2);
    
    -- Obtener datos actuales
    SELECT r.id_recurso, r.estado, rec.capacidad, rec.precio_por_hora 
    INTO v_id_recurso, v_estado, v_capacidad, v_precio_hora
    FROM Reservas r
    INNER JOIN Recursos rec ON r.id_recurso = rec.id_recurso
    WHERE r.id_reserva = p_id_reserva;
    
    IF v_estado IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La reserva no existe.';
    ELSEIF v_estado != 'Activa' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Solo se pueden modificar reservas activas.';
    END IF;

    IF p_nueva_cantidad_personas > v_capacidad THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La nueva cantidad de asistentes supera la capacidad del recurso.';
    END IF;

    IF p_nueva_hora_inicio >= p_nueva_hora_fin THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La hora de inicio debe ser anterior a la hora de fin.';
    END IF;

    IF p_nueva_fecha < CURDATE() OR (p_nueva_fecha = CURDATE() AND p_nueva_hora_inicio <= CURTIME()) THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'No se puede reprogramar una reserva para una fecha/hora pasada.';
    END IF;

    IF fn_verificar_disponibilidad(v_id_recurso, p_nueva_fecha, p_nueva_hora_inicio, p_nueva_hora_fin, p_id_reserva) = FALSE THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El nuevo horario se superpone con otra reserva existente.';
    END IF;
    
    SET v_horas = TIMESTAMPDIFF(MINUTE, p_nueva_hora_inicio, p_nueva_hora_fin) / 60.0;
    SET v_nuevo_costo = v_precio_hora * v_horas;

    UPDATE Reservas 
    SET fecha_reserva = p_nueva_fecha, 
        hora_inicio = p_nueva_hora_inicio, 
        hora_fin = p_nueva_hora_fin, 
        cantidad_personas = p_nueva_cantidad_personas,
        costo_total = v_nuevo_costo
    WHERE id_reserva = p_id_reserva;
END //
DELIMITER ;

-- =========================================================
-- PROCEDIMIENTO 3: Cancelar Reserva
-- =========================================================
DELIMITER //
CREATE PROCEDURE sp_cancelar_reserva(IN p_id_reserva INT)
BEGIN
    DECLARE v_estado VARCHAR(20);
    
    SELECT estado INTO v_estado FROM Reservas WHERE id_reserva = p_id_reserva;
    
    IF v_estado IS NULL THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La reserva seleccionada no existe.';
    ELSEIF v_estado = 'Cancelada' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'La reserva ya se encuentra cancelada.';
    ELSEIF v_estado = 'Finalizada' THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'No se puede cancelar una reserva que ya finalizó.';
    END IF;

    UPDATE Reservas SET estado = 'Cancelada' WHERE id_reserva = p_id_reserva;
END //

DELIMITER ;


-- =========================================================
-- 1. TABLA DE AUDITORÍA (Necesaria para el Trigger)
-- =========================================================
CREATE TABLE Auditoria_Cancelaciones (
    id_auditoria INT AUTO_INCREMENT PRIMARY KEY,
    id_reserva INT NOT NULL,
    id_cliente INT NOT NULL,
    fecha_cancelacion DATETIME DEFAULT CURRENT_TIMESTAMP,
    usuario_db VARCHAR(50)
);

-- =========================================================
-- 2. TRIGGER: Auditoría de Cancelaciones
-- =========================================================
DELIMITER //
CREATE TRIGGER trg_auditoria_cancelacion_reserva
AFTER UPDATE ON Reservas FOR EACH ROW
BEGIN
    -- Si el estado cambió y el nuevo estado es 'Cancelada'
    IF OLD.estado != 'Cancelada' AND NEW.estado = 'Cancelada' THEN
        INSERT INTO Auditoria_Cancelaciones (id_reserva, id_cliente, usuario_db)
        VALUES (OLD.id_reserva, OLD.id_cliente, CURRENT_USER());
    END IF;
END //
DELIMITER ;

-- VISTA 1: Detalle completo de reservas (Ideal para el DataGridView principal)
DELIMITER //
CREATE VIEW vw_detalle_reservas AS
SELECT 
    r.id_reserva, 
    r.fecha_reserva, 
    r.hora_inicio, 
    r.hora_fin, 
    r.cantidad_personas,
    r.costo_total, 
    r.estado AS estado_reserva,
    c.nombre AS cliente_nombre, 
    c.apellido AS cliente_apellido, 
    c.dni, 
    rec.nombre AS recurso_nombre, 
    t.descripcion AS tipo_recurso
FROM Reservas r
INNER JOIN Clientes c ON r.id_cliente = c.id_cliente
INNER JOIN Recursos rec ON r.id_recurso = rec.id_recurso
INNER JOIN TiposRecurso t ON rec.id_tipo = t.id_tipo;

DELIMITER ;

-- VISTA 2: Resumen estadístico de recursos 
DELIMITER //
CREATE VIEW vw_resumen_recursos AS
SELECT 
    rec.id_recurso,
    rec.nombre AS recurso_nombre,
    t.descripcion AS tipo,
    rec.estado,
    COUNT(CASE WHEN r.estado = 'Activa' THEN 1 END) AS reservas_activas,
    COUNT(r.id_reserva) AS total_reservas_historicas,
    IFNULL(SUM(CASE WHEN r.estado = 'Finalizada' THEN r.costo_total ELSE 0 END), 0) AS recaudacion_finalizada
FROM Recursos rec
INNER JOIN TiposRecurso t ON rec.id_tipo = t.id_tipo
LEFT JOIN Reservas r ON rec.id_recurso = r.id_recurso
GROUP BY rec.id_recurso, rec.nombre, t.descripcion, rec.estado;

DELIMITER ;


-- =========================================================
-- 1. TABLA: Adicionales elegidos en una reserva
-- =========================================================
CREATE TABLE Reservas_Adicionales (
    id_reserva INT,
    id_adicional INT,
    precio_cobrado DECIMAL(10,2), -- Congela el precio en caso de que aumente a futuro
    PRIMARY KEY (id_reserva, id_adicional),
    FOREIGN KEY (id_reserva) REFERENCES Reservas(id_reserva) ON DELETE CASCADE,
    FOREIGN KEY (id_adicional) REFERENCES Adicionales(id_adicional) ON DELETE CASCADE
);

DELIMITER //

-- =========================================================
-- 2. SP PARA LA INTERFAZ: Qué adicionales mostrar
-- =========================================================
-- Cuando el usuario selecciona un recurso en el ComboBox de C#, 
-- llamás a este SP para cargar los CheckBox de adicionales permitidos.
CREATE PROCEDURE sp_obtener_adicionales_por_recurso(IN p_id_recurso INT)
BEGIN
    SELECT a.id_adicional, a.descripcion, a.costo_extra
    FROM Adicionales a
    INNER JOIN Recursos r ON a.id_tipo_recurso = r.id_tipo
    WHERE r.id_recurso = p_id_recurso;
END //

DELIMITER ;
-- =========================================================
-- 3. SP PARA GUARDAR: Agregar el adicional a la reserva
-- =========================================================
DELIMITER //
CREATE PROCEDURE sp_agregar_adicional_reserva(
    IN p_id_reserva INT,
    IN p_id_adicional INT
)
BEGIN
    DECLARE v_id_tipo_recurso INT;
    DECLARE v_id_tipo_adicional INT;
    DECLARE v_costo_extra DECIMAL(10,2);
    
    -- Obtener a qué tipo pertenece el recurso que se reservó
    SELECT r.id_tipo INTO v_id_tipo_recurso
    FROM Reservas res
    INNER JOIN Recursos r ON res.id_recurso = r.id_recurso
    WHERE res.id_reserva = p_id_reserva;
    
    -- Obtener para qué tipo está diseñado el adicional y cuánto cuesta
    SELECT id_tipo_recurso, costo_extra INTO v_id_tipo_adicional, v_costo_extra
    FROM Adicionales WHERE id_adicional = p_id_adicional;
    
    -- BARRERA DE SEGURIDAD: Validar que el adicional corresponda al recurso
    IF v_id_tipo_recurso != v_id_tipo_adicional THEN
        SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'Operación denegada: El adicional seleccionado no está disponible para este tipo de recurso.';
    END IF;
    
    -- 1. Registrar el adicional
    INSERT INTO Reservas_Adicionales (id_reserva, id_adicional, precio_cobrado)
    VALUES (p_id_reserva, p_id_adicional, v_costo_extra);
    
    -- 2. Actualizar el costo total de la reserva sumando el valor del adicional
    UPDATE Reservas 
    SET costo_total = costo_total + v_costo_extra 
    WHERE id_reserva = p_id_reserva;
END //

DELIMITER ;
